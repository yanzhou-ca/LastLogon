    // ---------- helpers ----------------------------------------------------------------------

    private static LogonEventRecord Rec(
        string computer, string account, string? domain, int? logonType, int eventId = 4624,
        DateTimeOffset? time = null)
        => new(computer, time ?? new DateTimeOffset(2025, 3, 4, 5, 0, 0, TimeSpan.Zero), eventId,
            "evt", account, domain, null, logonType, logonType?.ToString(), null,
            new Dictionary<string, string>());

    private sealed class ListScanner(params LogonScanUpdate[] updates) : ILogonEventScanner
    {
        public async IAsyncEnumerable<LogonScanUpdate> ScanAsync(
            IEnumerable<string> computers, DateTimeOffset f, DateTimeOffset t,
            [EnumeratorCancellation] CancellationToken ct)
        {
            foreach (var u in updates) { yield return u; await Task.Yield(); }
        }
    }

    /// <summary>A scanner whose every call blocks on its own gate, like a slow remote query.</summary>
    private sealed class BlockingScanner : ILogonEventScanner
    {
        public readonly List<TaskCompletionSource> Gates = new();
        public readonly List<CancellationToken> Tokens = new();
        public readonly List<TaskCompletionSource> FirstEventSeen = new();

        public IAsyncEnumerable<LogonScanUpdate> ScanAsync(
            IEnumerable<string> computers, DateTimeOffset f, DateTimeOffset t, CancellationToken ct)
        {
            var gate = new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);
            var seen = new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);
            Gates.Add(gate); Tokens.Add(ct); FirstEventSeen.Add(seen);
            return Run(gate, seen, Gates.Count, ct);
        }

        private static async IAsyncEnumerable<LogonScanUpdate> Run(
            TaskCompletionSource gate, TaskCompletionSource seen, int n,
            [EnumeratorCancellation] CancellationToken ct)
        {
            yield return new LogonEventFound(Rec("pc01", "scan" + n, "CORP", 2));
            seen.TrySetResult();
            await gate.Task; // the real scanner only ends when its readers end
            yield return new LogonComputerCompleted("pc01", LogonQueryStatus.Success, null, 1);
        }
    }

    // ---------- R1  Enter key re-enters a running history scan --------------------------------
    // ScanWindow.LookupInput_PreviewKeyDown calls StartScanAsync() without checking IsScanning.

    [Fact]
    public async Task R1_History_SecondScanWhileRunning_CancelsTheFirstOne()
    {
        var scanner = new BlockingScanner();
        var vm = new LogonHistoryViewModel(scanner) { ComputersText = "pc01" };

        var scan1 = vm.ScanAsync();
        await scanner.FirstEventSeen[0].Task.WaitAsync(TimeSpan.FromSeconds(5));
        var scan2 = vm.ScanAsync();                       // what pressing Enter again does
        await scanner.FirstEventSeen[1].Task.WaitAsync(TimeSpan.FromSeconds(5));

        _out.WriteLine($"scan1 token cancelled after scan2 started: {scanner.Tokens[0].IsCancellationRequested}");
        vm.Cancel();                                      // operator presses Cancel
        _out.WriteLine($"scan1 token cancelled after Cancel(): {scanner.Tokens[0].IsCancellationRequested}");
        _out.WriteLine($"scan2 token cancelled after Cancel(): {scanner.Tokens[1].IsCancellationRequested}");

        scanner.Gates[0].SetResult(); scanner.Gates[1].SetResult();
        await Task.WhenAll(scan1, scan2).WaitAsync(TimeSpan.FromSeconds(5));

        Assert.True(scanner.Tokens[0].IsCancellationRequested,
            "the first scan keeps running and can no longer be cancelled");
    }

    [Fact]
    public async Task R1_History_IsScanningStaysTrueWhileTheSecondScanRuns()
    {
        var scanner = new BlockingScanner();
        var vm = new LogonHistoryViewModel(scanner) { ComputersText = "pc01" };

        var scan1 = vm.ScanAsync();
        await scanner.FirstEventSeen[0].Task.WaitAsync(TimeSpan.FromSeconds(5));
        var scan2 = vm.ScanAsync();
        await scanner.FirstEventSeen[1].Task.WaitAsync(TimeSpan.FromSeconds(5));

        scanner.Gates[0].SetResult();                     // first scan ends, second still blocked
        await scan1.WaitAsync(TimeSpan.FromSeconds(5));
        var stillScanning = vm.IsScanning;
        _out.WriteLine($"IsScanning after scan1 finished, scan2 still running: {stillScanning}");

        scanner.Gates[1].SetResult();
        await scan2.WaitAsync(TimeSpan.FromSeconds(5));
        Assert.True(stillScanning, "Scan button would be re-enabled while a scan is still running");
    }

    // ---------- R2  same for the profile scan ---------------------------------------------------

    private sealed class SlowFs : IProfileFileSystem
    {
        public TimeSpan ListDelay = TimeSpan.Zero;
        public TimeSpan ProfileDelay = TimeSpan.Zero;
        public int Profiles = 1;
        public Func<string, Exception?>? FailProfile;
        public IReadOnlyList<string> ListUserFolders(string c)
        {
            Thread.Sleep(ListDelay);
            return Enumerable.Range(0, Profiles).Select(i => "user" + i).ToList();
        }
        public DateTime? GetNtUserDatTime(string c, string u)
        {
            Thread.Sleep(ProfileDelay);
            var ex = FailProfile?.Invoke(u); if (ex != null) throw ex;
            return new DateTime(2025, 1, 1);
        }
        public DateTime GetFolderTime(string c, string u) => new(2020, 1, 1);
    }

    [Fact]
    public async Task R2_Profile_IsScanningStaysTrueWhileTheSecondScanRuns()
    {
        var fs = new SlowFs { ListDelay = TimeSpan.FromSeconds(2) };
        var vm = new MainViewModel(new ProfileScanner(fs, 4, TimeSpan.FromSeconds(30))) { ComputersText = "pc1" };

        var scan1 = vm.ScanAsync();
        await Task.Delay(200);
        var scan2 = vm.ScanAsync();                       // Enter pressed again
        await scan1.WaitAsync(TimeSpan.FromSeconds(5));   // scan1 unwinds as "cancelled"
        var stillScanning = vm.IsScanning;
        _out.WriteLine($"IsScanning after scan1 unwound while scan2 runs: {stillScanning}; status='{vm.StatusText}'");
        await scan2.WaitAsync(TimeSpan.FromSeconds(10));
        Assert.True(stillScanning);
    }

    // ---------- R3  one unreadable profile discards the whole computer --------------------------

    [Fact]
    public async Task R3_OneProfileWithDeniedAcl_DoesNotHideTheOtherProfiles()

    }

    [Fact]
    public async Task R5_LocalFilter_WorksWhenTheOperatorTypedAFullyQualifiedName()
    {
        // Windows records the NetBIOS machine name (PC01) as the domain of a local account.
        var local = Rec("pc01.corp.example", "localadmin", "PC01", 2);

    [Fact]

    // ---------- R9  event parser robustness -----------------------------------------------------

    private static string EventXml(string eventId, string logonType)
        => $"<Event xmlns='http://schemas.microsoft.com/win/2004/08/events/event'><System><EventID>{eventId}</EventID>" +
           "<TimeCreated SystemTime='2025-03-04T05:00:00.0000000Z'/></System><EventData>" +
           "<Data Name='TargetUserName'>alice</Data><Data Name='TargetDomainName'>CORP</Data>" +
           $"<Data Name='LogonType'>{logonType}</Data></EventData></Event>";

    private sealed class XmlSource(params string[] xml) : ILogonEventSource
    {
        public IEnumerable<string> ReadEventXml(string c, DateTimeOffset f, DateTimeOffset t, CancellationToken ct) => xml;
    }

    [Theory]
    [InlineData("abc", "2")]                 // FormatException  -> skipped (handled)
    [InlineData("99999999999", "2")]         // OverflowException -> ?
    [InlineData("4624", "99999999999")]      // OverflowException -> ?
    public async Task R9_OneBadRecord_MustNotFailTheWholeComputer(string eventId, string logonType)
    {
        var source = new XmlSource(EventXml("4624", "2"), EventXml(eventId, logonType), EventXml("4624", "3"));
        var updates = new List<LogonScanUpdate>();
        await foreach (var u in new LogonEventScanner(source).ScanAsync(new[] { "pc01" }, DateTimeOffset.UnixEpoch, DateTimeOffset.UtcNow, default))
            updates.Add(u);
        var done = updates.OfType<LogonComputerCompleted>().Single();
        _out.WriteLine($"eventId={eventId} logonType={logonType}: status={done.Status} events={done.EventCount} msg={done.Message}");
        Assert.Equal(LogonQueryStatus.Success, done.Status);
        Assert.Equal(2, done.EventCount);
    }

    // ---------- R10  path-segment guard ---------------------------------------------------------

    [Theory]
    [InlineData(".. ")]
    [InlineData("..\t")]
    [InlineData(" .. ")]
    [InlineData("...")]
    [InlineData(". ")]
    public void R10_ProfileFolderGuard_RejectsNamesWindowsNormalisesIntoDotSegments(string name)
    {
        var path = Path.Combine(@"\\pc01.corp.example\C$\Users", name, "NTUSER.DAT");
        var full = Path.GetFullPath(path);
        _out.WriteLine($"guard says safe={InputParser.IsSafeProfileFolderName(name)} for '{name}', GetFullPath => {full}");
        var normalisedAwayFromUsers = !full.StartsWith(@"\\pc01.corp.example\C$\Users\", StringComparison.OrdinalIgnoreCase)
                                      || full.EndsWith(@"\Users\NTUSER.DAT", StringComparison.OrdinalIgnoreCase);
        Assert.False(InputParser.IsSafeProfileFolderName(name) && normalisedAwayFromUsers,
            "guard accepts a name that collapses to a dot segment");
    }

    // ---------- R13  CSV / TSV exporter ---------------------------------------------------------

    [Fact]
    [Trait("kind", "characterization")]
    public void R13_Export_FormulaAndNumericEdgeCases()
    {
        ProfileResult R(string user, DateTime? t, string? msg = null) => new("pc01", user, t, LastLogon.Core.ActivitySource.NtUserDat, ScanStatus.Ok, msg);
        var rows = new[]
        {
            R("=1+1", DateTime.Now), R("\t=1+1", DateTime.Now), R("-bob", DateTime.Now), R("@SUM(A1)", DateTime.Now),
            R("ok", DateTime.Now.AddDays(3)),                           // clock skew: future timestamp -> negative DaysAgo
            R("line\r\nbreak,\"q\"", DateTime.Now),
            R("\u00A0=1+1", DateTime.Now),                                // NBSP then '='
            R("\uFF1D1+1", DateTime.Now),                                 // full-width '='
        };
        _out.WriteLine(ResultExporter.ToCsv(rows));
        _out.WriteLine("--- TSV ---");
        _out.WriteLine(ResultExporter.ToTabSeparated(rows));
    }

public class ReproTests2
{
    private readonly ITestOutputHelper _out;
    public ReproTests2(ITestOutputHelper output) => _out = output;

    [Theory]
    [InlineData("th-TH")]   // Thai Buddhist calendar is the default calendar of this culture
    [InlineData("da-DK")]   // time separator is '.'
    [InlineData("ar-SA")]   // Umm al-Qura calendar by default
    public void R15_ProfileCsv_TimestampDoesNotDependOnTheOperatorsCulture(string culture)
    {
        var old = System.Globalization.CultureInfo.CurrentCulture;
        try
        {
            System.Globalization.CultureInfo.CurrentCulture = new System.Globalization.CultureInfo(culture);
            var row = new ProfileResult("pc", "u", new DateTime(2025, 3, 4, 15, 30, 0), LastLogon.Core.ActivitySource.NtUserDat, ScanStatus.Ok, null);
            var cell = ResultExporter.ToCsv(new[] { row }).Split("\r\n")[1].Split(',')[2];
            var evtCsv = ResultExporter.ToLogonEventCsv(new[] { new LogonEventRecord("pc", new DateTimeOffset(2025, 3, 4, 15, 30, 0, TimeSpan.Zero), 4624, "x", "u", "d", null, 2, "Interactive", null, new Dictionary<string, string>()) });
            _out.WriteLine($"{culture}: profile CSV time = '{cell}'   |   LastActivityText = '{row.LastActivityText}'   |   logon-event CSV LocalTime = '{evtCsv.Split("\r\n")[1].Split(',')[1]}'");
            Assert.Equal("2025-03-04 15:30", cell);
        }
        finally { System.Globalization.CultureInfo.CurrentCulture = old; }
    }
